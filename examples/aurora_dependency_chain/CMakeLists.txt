cmake_minimum_required(VERSION 3.17)
project(aurora_dependency_chain LANGUAGES CXX)
set(GODOTCPP_API_VERSION "4.7" CACHE STRING "")
set(GODOTCPP_AURORAOS ON CACHE BOOL "")
set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/profile.json" CACHE FILEPATH "")
add_subdirectory(../.. godot-cpp)
include(../../cmake/auroraos.cmake)
add_library(aurora_probe_leaf SHARED leaf.cpp)
add_library(aurora_probe_middle SHARED middle.cpp)
target_link_libraries(aurora_probe_middle PRIVATE aurora_probe_leaf)
add_library(aurora_probe SHARED extension.cpp)
target_link_libraries(aurora_probe PRIVATE godot-cpp aurora_probe_middle)
foreach(target aurora_probe_leaf aurora_probe_middle aurora_probe)
    auroraos_private_library(${target})
    install(TARGETS ${target} LIBRARY DESTINATION lib)
endforeach()
