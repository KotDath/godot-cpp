cmake_minimum_required(VERSION 3.17)
project(aurora_hot_reload LANGUAGES CXX)
set(GODOTCPP_API_VERSION "4.7" CACHE STRING "")
set(GODOTCPP_AURORAOS ON CACHE BOOL "")
set(GODOTCPP_USE_HOT_RELOAD ON CACHE BOOL "")
set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/profile.json" CACHE FILEPATH "")
set(AURORA_RELOAD_REVISION "1" CACHE STRING "Fixture code revision, 1 or 2")
if(NOT AURORA_RELOAD_REVISION MATCHES "^[12]$")
    message(FATAL_ERROR "AURORA_RELOAD_REVISION must be 1 or 2")
endif()
add_subdirectory(../.. godot-cpp)
include(../../cmake/auroraos.cmake)
add_library(aurora_reload SHARED extension.cpp)
target_compile_definitions(aurora_reload PRIVATE AURORA_RELOAD_REVISION=${AURORA_RELOAD_REVISION})
target_link_libraries(aurora_reload PRIVATE godot-cpp)
auroraos_private_library(aurora_reload)
